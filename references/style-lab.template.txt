function exportCss() {
  const v = (m) => { const old = S.mode; S.mode = m; const o = cssVars(); S.mode = old; return Object.entries(o).filter(([k]) => !k.startsWith('--clay') && !k.startsWith('--glass')).map(([k, x]) => `  ${k.replace('--', '--luz-')}: ${x};`).join('\n') }
  return `:root {\n${v('light')}\n}\n@media (prefers-color-scheme: dark) {\n  :root:not([data-theme="light"]) {\n${v('dark').replace(/^/gm, '  ')}\n  }\n}\n:root[data-theme="dark"] {\n${v('dark')}\n}`
}
function exportJson() {
  const L = tokens('light'), D = tokens('dark'), n = { base: 'action', press: 'press', tint: 'tint', on: 'on', text: 'text' }
  const c = (l, d) => ({ $type: 'color', $value: l, $extensions: { mode: { dark: d } } })
  const out = {
    $description: `风格实验室导出：配色 ${DATA.themes[S.palette].label}（${S.palette}）`,
    color: { brand: Object.fromEntries(Object.entries(n).map(([k, key]) => [k, c(L[key], D[key])])), accent: { base: c(L.accent, D.accent), tint: c(L.accentTint, D.accentTint) }, surface: { canvas: c(L.canvas, D.canvas), card: c(L.surface, D.surface), nav: c(L.nav, D.nav) }, ink: { primary: c(L.ink, D.ink), secondary: c(L.ink2, D.ink2), tertiary: c(L.ink3, D.ink3) }, line: c(L.line, D.line), price: c(L.price, D.price) },
    style: { icons: S.icons, shape: S.shape, depth: S.depth, type: S.type, motion: S.motion, glassOnNavigation: S.glassNav },
  }
  return JSON.stringify(out, null, 2)
}
function exportBrief() {
  const T = DATA.themes[S.palette]
  return `# 风格方案\n\n- 配色：${T.label}（${S.palette}），${S.mode === 'light' ? '浅色' : '深色'}为当前查看\n- 图标：${OPTLabel('icons')}（控件图标始终是 2px 线性）\n- 圆角：${OPTLabel('shape')}\n- 层级：${OPTLabel('depth')}${S.glassNav ? '，导航与标签栏用毛玻璃（待裁定）' : ''}\n- 字阶：${OPTLabel('type')}\n- 动效：${OPTLabel('motion')}\n\n下一步：把它写进 BRIEF；颜色取 themes.json 的 ${S.palette}；运行 audit_theme_discipline.py --combo；待裁定项见页面右侧。`
}

/* ===== 渲染 ===== */
function seg(key) { return `<div class="seg" role="group" aria-label="${LBL[key]}">${OPT[key].map(([v, t, flag]) => `<button type="button" data-k="${key}" data-v="${v}" aria-pressed="${S[key] === v}">${t}${flag ? '<span class="flag">待裁定</span>' : ''}</button>`).join('')}</div>` }
function controls() {
  const pre = Object.entries(PRESETS).map(([id, p]) => `<button type="button" class="pre" data-pre="${id}" aria-pressed="${S.preset === id}"><b>${id} · ${p.name}</b><span>${p.desc}</span></button>`).join('')
  const CUR = ['florist', 'b2b-graphite', 'travel-azure', 'fitness-gym', 'wine-jewel', 'agri-green', 'pet', 'ai-agent', 'coffee', 'western-light'].filter((id) => DATA.themes[id])
  const pal = CUR.map((id) => { const t = DATA.themes[id]; return `<button type="button" class="chip" data-pal="${id}" aria-pressed="${S.palette === id}"><i style="background:${t.action}"></i>${t.label}</button>` }).join('')
  const rest = Object.entries(DATA.themes).filter(([id]) => !CUR.includes(id))
  const more = rest.length ? `<select id="more" class="in" aria-label="更多配色" style="margin-top:8px;width:100%"><option value="">更多配色（${rest.length} 套）…</option>${rest.map(([id, t]) => `<option value="${id}" ${S.palette === id ? 'selected' : ''}>${t.label}（${t.industry}）</option>`).join('')}</select>` : ''
  return `<h2>三套预设</h2><div class="row">${pre}</div>
  <span class="lbl">配色（${Object.keys(DATA.themes).length} 套，来自主题库）</span><div class="row">${pal}</div>${more}
  ${['mode', 'icons', 'shape', 'depth', 'type', 'motion'].map((k) => `<span class="lbl">${LBL[k]}</span>${seg(k)}`).join('')}
  <span class="lbl">毛玻璃的使用范围</span><div class="seg"><button type="button" data-glassnav="0" aria-pressed="${!S.glassNav}">仅主视觉装饰</button><button type="button" data-glassnav="1" aria-pressed="${S.glassNav}">含导航栏与标签栏<span class="flag">待裁定</span></button></div>`
}
function render() {
  const pv = document.getElementById('pv')
  const vars = cssVars(); Object.entries(vars).forEach(([k, v]) => pv.style.setProperty(k, v))
  pv.className = `pv ty-${S.type} mo-${S.motion} sh-${S.shape} dp-${S.depth} ic-${S.icons === 'duo' ? 'duo' : S.icons === 'clay' ? 'clay' : 'line'} m-${S.mode}`
  pv.innerHTML = S.view === 'mall' ? mall() : S.view === 'admin' ? admin() : kit()
  document.getElementById('controls').innerHTML = controls()
  document.getElementById('checks').innerHTML = checks()
  document.querySelectorAll('#tabs .tab').forEach((b) => b.setAttribute('aria-selected', b.dataset.v === S.view))
  const demo = document.getElementById('demo'); if (demo) demo.onclick = () => { demo.classList.toggle('go') }
}
document.addEventListener('change', (e) => { if (e.target.id === 'more' && e.target.value) { S.palette = e.target.value; S.preset = ''; render() } })
document.addEventListener('click', (e) => {
  const t = e.target.closest('button'); if (!t) return
  if (t.dataset.pre) { S = { ...PRESETS[t.dataset.pre], preset: t.dataset.pre, view: S.view }; render() }
  else if (t.dataset.pal) { S.palette = t.dataset.pal; S.preset = ''; render() }
  else if (t.dataset.k) { S[t.dataset.k] = t.dataset.v; S.preset = ''; render() }
  else if (t.dataset.glassnav !== undefined) { S.glassNav = t.dataset.glassnav === '1'; S.preset = ''; render() }
  else if (t.dataset.v && t.closest('#tabs')) { S.view = t.dataset.v; render() }
  else if (t.dataset.ex) { const o = document.getElementById('exout'); o.value = t.dataset.ex === 'css' ? exportCss() : t.dataset.ex === 'json' ? exportJson() : exportBrief(); o.focus(); o.select() }
})
document.getElementById('export').innerHTML = `<h2>导出</h2><div class="row"><button class="btn" data-ex="css" type="button">CSS 变量</button><button class="btn" data-ex="json" type="button">令牌 JSON</button><button class="btn" data-ex="brief" type="button">风格简报</button></div><textarea id="exout" class="out" aria-label="导出内容" placeholder="点上面的按钮，这里出现可复制的内容"></textarea><p class="note">颜色一律取自主题库；图标、圆角、层级、字阶、动效是选项，规范外的会标“待裁定”。</p>`
render()
</script>
